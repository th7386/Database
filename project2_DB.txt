drop table if exists course;
drop table if exists favorite_course;
drop table if exists favorite_food;
drop table if exists planguage; 
drop table if exists preferred_pl; 
drop table if exists professor; 
drop table if exists student;


drop table if exists my_board;
create table my_board (id INT auto_increment, nick_name varchar(100),
mytext varchar(1000), primary key(id));


CREATE TABLE course (
Course_ID varchar(100),Course_Name varchar(100),Offering_Department varchar(100),Maximum_capacity varchar(100),Prof_ID varchar(100),  primary key(course_id));


CREATE TABLE favorite_course (
Nick_name varchar(100), Course_name varchar(100),  primary key(Nick_name, Course_name));


CREATE TABLE favorite_food (
Nick_name varchar(100), Food varchar(100),  primary key(Nick_name, Food));

CREATE TABLE planguage (
Name varchar(100), Developer varchar(100), Market_ratio varchar(100), primary key(name));


CREATE TABLE preferred_pl (
Nick_name varchar(100), PName varchar(100), primary key(Nick_name, Pname));


CREATE TABLE professor (
Prof_ID varchar(100), SSN varchar(100),Name varchar(100),Major varchar(100), Lab_name varchar(100), primary key(Prof_ID));


CREATE TABLE student(Nick_name varchar(100), Section varchar(100), Gender varchar(100), MBTI varchar(100), Workout_duration varchar(100), Sleep_duration varchar(100), Dream varchar(100), PRIMARY KEY(Nick_Name));
